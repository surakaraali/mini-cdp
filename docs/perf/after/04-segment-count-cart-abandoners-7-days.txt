-- Segment count: cart abandoners, 7 days
-- args: [add_to_cart now - 168h0m0s 1 purchase now - 168h0m0s 0]

WITH ec AS (
    SELECT user_id,
           count(*) FILTER (WHERE name = $1 AND occurred_at >= $2) AS c0,
           count(*) FILTER (WHERE name = $4 AND occurred_at >= $5) AS c1
    FROM events
    WHERE (name = $1 AND occurred_at >= $2) OR (name = $4 AND occurred_at >= $5)
    GROUP BY user_id
)
SELECT count(*)
FROM users u LEFT JOIN ec ON ec.user_id = u.id
WHERE (COALESCE(ec.c0, 0) >= $3) AND (COALESCE(ec.c1, 0) <= $6)
