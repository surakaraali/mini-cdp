-- args: [purchase 0]

WITH ec AS (
    SELECT user_id,
           count(*) FILTER (WHERE name = $1) AS c0
    FROM events
    WHERE (name = $1)
    GROUP BY user_id
)
SELECT count(*)
FROM users u LEFT JOIN ec ON ec.user_id = u.id
WHERE (COALESCE(ec.c0, 0) <= $2)
